cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(octopus VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OCTO_ASAN "Enable AddressSanitizer" OFF)
set(OCTO_GODOT_CPP_DIR "" CACHE PATH "Verified, locked godot-cpp source checkout")
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
add_library(octo_core STATIC core/src/world.cpp core/src/dynamics.cpp core/src/checkpoint.cpp core/src/contact.cpp core/src/broadphase.cpp core/src/constraints.cpp core/src/coupled.cpp core/src/scene.cpp core/src/torso.cpp)
add_library(octo_fem STATIC core/src/fem.cpp)
target_include_directories(octo_fem PUBLIC core/include)
target_link_libraries(octo_core PUBLIC octo_fem)
target_compile_features(octo_fem PUBLIC cxx_std_20)
if(MSVC)
 target_compile_options(octo_fem PUBLIC /W4 /WX /permissive- /utf-8 /fp:precise)
 target_link_options(octo_fem PUBLIC /STACK:8388608)
 if(OCTO_ASAN)
  target_compile_options(octo_fem PUBLIC /fsanitize=address /Z7)
  target_link_options(octo_fem PUBLIC /INCREMENTAL:NO)
 endif()
endif()
add_executable(octo_fem_tests tests/unit/fem_tests.cpp)
target_link_libraries(octo_fem_tests PRIVATE octo_fem)
file(GLOB_RECURSE OCTO_BUILD_INPUTS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/core/*.cpp" "${CMAKE_SOURCE_DIR}/core/*.hpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${OCTO_BUILD_INPUTS})
file(SHA256 "${CMAKE_SOURCE_DIR}/CMakeLists.txt" OCTO_BUILD_CONFIG_SHA)
set(OCTO_SOURCE_FINGERPRINT "${OCTO_BUILD_CONFIG_SHA}-${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}-${CMAKE_BUILD_TYPE}-${CMAKE_SYSTEM_PROCESSOR}-${OCTO_ASAN}-${CMAKE_CXX_FLAGS}-${CMAKE_CXX_FLAGS_DEBUG}-${CMAKE_CXX_FLAGS_RELEASE}")
foreach(source_file IN LISTS OCTO_BUILD_INPUTS)
 file(SHA256 "${source_file}" source_sha)
 string(APPEND OCTO_SOURCE_FINGERPRINT "${source_sha}")
endforeach()
string(SHA256 OCTO_CORE_BUILD_ID "${OCTO_SOURCE_FINGERPRINT}")
target_compile_definitions(octo_core PUBLIC OCTO_BUILD_ID="${OCTO_CORE_BUILD_ID}")
target_include_directories(octo_core PUBLIC core/include)
set_target_properties(octo_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_link_options(octo_core PUBLIC /STACK:8388608)
  target_compile_options(octo_core PUBLIC /W4 /WX /permissive- /utf-8 /fp:precise)
  if(OCTO_ASAN)
    target_compile_options(octo_core PUBLIC /fsanitize=address /Z7)
    target_link_options(octo_core PUBLIC /INCREMENTAL:NO)
  endif()
else()
  target_compile_options(octo_core PUBLIC -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  if(OCTO_ASAN)
    target_compile_options(octo_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(octo_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(octo_cli core/src/cli.cpp)
target_link_libraries(octo_cli PRIVATE octo_core)
add_executable(octo_sim core/src/sim_cli.cpp)
target_link_libraries(octo_sim PRIVATE octo_core)
add_executable(octo_replay core/src/replay_cli.cpp)
target_link_libraries(octo_replay PRIVATE octo_core)
add_executable(octo_contact_regression tools/repro_live_step.cpp)
target_link_libraries(octo_contact_regression PRIVATE octo_core)
add_executable(octo_live godot_bridge/src/live_server.cpp)
target_link_libraries(octo_live PRIVATE octo_core)
add_executable(octo_benchmark tools/native_benchmark.cpp)
target_link_libraries(octo_benchmark PRIVATE octo_core)
add_executable(octo_core_tests tests/unit/world_tests.cpp)
target_link_libraries(octo_core_tests PRIVATE octo_core)
enable_testing()
add_test(NAME physics.live_wrap_regression COMMAND octo_contact_regression "${CMAKE_SOURCE_DIR}/tests/scenes/regressions/live_wrap_319.octostate")
set_tests_properties(physics.live_wrap_regression PROPERTIES TIMEOUT 600)
add_test(NAME live.protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/live_bridge_tests.py $<TARGET_FILE:octo_live>)
set_tests_properties(live.protocol PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 240 LABELS "long")
add_test(NAME live.invalid_request_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/live_invalid_protocol_tests.py $<TARGET_FILE:octo_live>)
set_tests_properties(live.invalid_request_recovery PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 60)
add_test(NAME FEM.P1V1 COMMAND octo_fem_tests)
add_executable(octo_p2_tests tests/unit/p2_tests.cpp)
target_link_libraries(octo_p2_tests PRIVATE octo_core)
target_compile_definitions(octo_p2_tests PRIVATE OCTO_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
foreach(id V06_geometry V07 V08 door ccd V10 capsules V12_io hessian)
 add_test(NAME P2.${id} COMMAND octo_p2_tests ${id})
 set_tests_properties(P2.${id} PROPERTIES TIMEOUT 1800)
endforeach()
set_tests_properties(P2.V10 P2.V12_io PROPERTIES LABELS "long")
add_executable(octo_physics_tests tests/unit/physics_tests.cpp)
target_link_libraries(octo_physics_tests PRIVATE octo_core)
foreach(test_id math V01 V02 V03 V04 V05 V06 V07 V08_model V09_pair V09_rod V10_rest contact V11 V12 rollback implicit_exact free_spin)
  add_test(NAME physics.${test_id} COMMAND octo_physics_tests ${test_id})
  set_tests_properties(physics.${test_id} PROPERTIES TIMEOUT 180)
endforeach()
set_tests_properties(physics.V03 physics.V04 physics.V05 physics.V09_rod physics.V10_rest physics.V12 PROPERTIES LABELS "long")
add_test(NAME P0.core_contract COMMAND octo_core_tests)
add_test(NAME P0.cli_tick COMMAND octo_cli --ticks 1 --seed 42)
set_tests_properties(P0.cli_tick PROPERTIES PASS_REGULAR_EXPRESSION "\"tick\":1")
add_test(NAME P0.cli_reject_negative COMMAND octo_cli --ticks -1)
set_tests_properties(P0.cli_reject_negative PROPERTIES WILL_FAIL TRUE)
add_test(NAME P2.schemas COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/p2_schema_tests.py)
add_test(NAME P0.schemas COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/schema_tests.py)
add_test(NAME P0.core_engine_boundary COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/check_boundary.py)
if(OCTO_GODOT_CPP_DIR)
  add_subdirectory("${OCTO_GODOT_CPP_DIR}" "${CMAKE_BINARY_DIR}/godot-cpp")
  add_library(octo_bridge SHARED godot_bridge/src/bridge.cpp)
  target_link_libraries(octo_bridge PRIVATE octo_core godot-cpp)
  set_target_properties(octo_bridge PROPERTIES PREFIX "" OUTPUT_NAME "octo_bridge"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/game/bin"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/game/bin")
endif()
